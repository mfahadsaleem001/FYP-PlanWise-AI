const API_BASE_URL = "http://127.0.0.1:8000";

// Common API response handler
async function handleResponse(response, defaultMessage) {
  if (response.status === 204) {
    return null;
  }

  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    if (response.status === 401) {
      throw new Error(
        "Your session has expired. Please sign in again."
      );
    }

    if (response.status === 403) {
      throw new Error(
        "You are not authorized to perform this action."
      );
    }

    const detail = data?.detail;
    let message = "";

    if (typeof detail === "string") {
      message = detail;
    } else if (Array.isArray(detail)) {
      message = detail
        .map((item) => {
          if (typeof item === "string") {
            return item;
          }

          if (item && typeof item === "object") {
            const field = Array.isArray(item.loc)
              ? item.loc
                  .filter((part) => part !== "body")
                  .join(".")
              : "";

            const description =
              item.msg || item.message || "";

            if (field && description) {
              return `${field}: ${description}`;
            }

            return description || JSON.stringify(item);
          }

          return String(item);
        })
        .filter(Boolean)
        .join(", ");
    } else if (detail && typeof detail === "object") {
      message =
        detail.message ||
        detail.msg ||
        JSON.stringify(detail);
    } else if (typeof data?.message === "string") {
      message = data.message;
    } else if (typeof data?.error === "string") {
      message = data.error;
    }

    throw new Error(message || defaultMessage);
  }

  return data;
}

// Common API request helper
async function apiRequest(
  endpoint,
  {
    method = "GET",
    body,
    authenticated = true,
  } = {}
) {
  const headers = {};

  if (body !== undefined) {
    headers["Content-Type"] = "application/json";
  }

  if (authenticated) {
    const token = localStorage.getItem(
      "planwise_access_token"
    );

    if (!token) {
      throw new Error(
        "Admin authentication token not found. Please sign in again."
      );
    }

    headers.Authorization = `Bearer ${token}`;
  }

  const response = await fetch(
    `${API_BASE_URL}${endpoint}`,
    {
      method,
      headers,
      ...(body !== undefined
        ? { body: JSON.stringify(body) }
        : {}),
    }
  );

  return response;
}

// Extract project arrays from supported response shapes.
function extractProjects(response) {
  if (Array.isArray(response)) {
    return response;
  }

  if (Array.isArray(response?.items)) {
    return response.items;
  }

  if (Array.isArray(response?.data)) {
    return response.data;
  }

  if (Array.isArray(response?.results)) {
    return response.results;
  }

  if (Array.isArray(response?.projects)) {
    return response.projects;
  }

  if (Array.isArray(response?.data?.items)) {
    return response.data.items;
  }

  if (Array.isArray(response?.data?.projects)) {
    return response.data.projects;
  }

  return null;
}

// GET: Fetch all projects for the admin Tasks page.
// The primary endpoint remains unchanged. A fallback is attempted
// only when the primary endpoint returns an empty project list.
export async function getAdminProjects() {
  const response = await apiRequest(
    "/api/projects/admin/all"
  );

  const data = await handleResponse(
    response,
    "Failed to load projects."
  );

  const projects = extractProjects(data);

  if (projects === null) {
    throw new Error(
      "The projects API returned an unsupported response format."
    );
  }

  if (projects.length > 0) {
    return projects;
  }

  // Fallback: requires GET /api/projects to exist in the backend.
  const fallbackResponse = await apiRequest("/api/projects");

  const fallbackData = await handleResponse(
    fallbackResponse,
    "Failed to load projects from the fallback endpoint."
  );

  const fallbackProjects = extractProjects(fallbackData);

  if (fallbackProjects === null) {
    throw new Error(
      "The fallback projects API returned an unsupported response format."
    );
  }

  return fallbackProjects;
}

    `/api/admin/tasks${query ? `?${query}` : ""}`

    response,
    "Failed to load tasks."
  );
}

function formatDate(value) {
  if (!value) return "Not available";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return String(value);

  return date.toLocaleDateString("en-GB", {
    day: "2-digit",
    month: "short",
    year: "numeric",
  });
}

function formatStatus(status) {
  if (!status) return "Unknown";

  return String(status)
    .trim()
    .toLowerCase()
    .replace(/[_-]+/g, " ")
    .replace(/\b\w/g, (char) => char.toUpperCase());
}

function formatProjectType(type) {
  if (!type) return "Not available";

  return String(type)
    .trim()
    .toLowerCase()
    .replace(/[_-]+/g, " ")
    .replace(/\b\w/g, (char) => char.toUpperCase());
}

function getProjectOwner(project) {
  return (
    project.owner ||
    project.owner_name ||
    project.owner_email ||
    (project.owner_id ? `User #${project.owner_id}` : "Not available")
  );
}

function getCost(project) {
  const value =
    project.estimated_cost ??
    project.estimatedCost ??
    project.total_cost ??
    project.cost;

  if (value === null || value === undefined || value === "") {
    return "Not available";
  }

  if (typeof value === "string") return value;

  return new Intl.NumberFormat("en-US", {
    maximumFractionDigits: 2,
  }).format(value);
}

function getDuration(project) {
  const value =
    project.estimated_duration ??
    project.estimatedDuration ??
    project.duration_weeks ??
    project.duration;

  if (value === null || value === undefined || value === "") {
    return "Not available";
  }

  if (typeof value === "number") {
    return `${value} ${value === 1 ? "Week" : "Weeks"}`;
  }

  return String(value);
}

function getProgress(project) {
  const value =
    project.progress ??
    project.progress_percentage ??
    project.progressPercentage;

  if (value === null || value === undefined || value === "") {
    return null;
  }

  const number = Number(value);

  if (!Number.isFinite(number)) return null;

  return Math.min(100, Math.max(0, number));
}

function getRisk(project) {
  const risk =
    project.risk_level ??
    project.riskLevel ??
    project.risk;

  if (!risk) return null;

  if (typeof risk === "object") {
    return risk.severity || risk.level || risk.status || null;
  }

  return String(risk);
}

function getTeam(project) {
  const team =
    project.team ??
    project.assigned_resources ??
    project.assignedResources ??
    project.members;

  return Array.isArray(team) ? team : [];
}

function getTeamSize(project) {
  const explicitSize =
    project.team_size ??
    project.teamSize ??
    project.required_team_size;

  if (
    explicitSize !== null &&
    explicitSize !== undefined &&
    explicitSize !== ""
  ) {
    const number = Number(explicitSize);
    return Number.isFinite(number) ? number : null;
  }

  const team = getTeam(project);

  return team.length > 0 ? team.length : null;
}

function normalizeProject(project) {
  return {
    ...project,
    id: project.id,
    name: project.name || "Untitled Project",
    owner: getProjectOwner(project),
    type: formatProjectType(project.type || project.project_type),
    status: formatStatus(project.status),
    cost: getCost(project),
    duration: getDuration(project),
    progress: getProgress(project),
    team: getTeam(project),
    teamSize: getTeamSize(project),
    risk: getRisk(project),
    created: formatDate(
      project.created_at || project.createdAt || project.created
    ),
    deadline: formatDate(
      project.deadline || project.due_date || project.dueDate
    ),
    description:
      project.description || "No project description available.",
  };
}

function getStatusClass(status) {
  const normalized = String(status || "").toLowerCase();

  if (normalized === "completed") return "completed";

  if (
    normalized === "in development" ||
    normalized === "in progress" ||
    normalized === "active"
  ) {
    return "in-development";
  }

  if (normalized === "planning" || normalized === "draft") {
    return "planning";
  }

  if (normalized === "on hold" || normalized === "paused") {
    return "on-hold";
  }

  return "";
}

function getRiskClass(risk) {
  const normalized = String(risk || "").toLowerCase();

  if (normalized.includes("critical")) return "high";
  if (normalized.includes("high")) return "high";
  if (normalized.includes("medium")) return "medium";
  if (normalized.includes("low")) return "low";

  return "";
}

}

      } else {

      }
    }
  };

  const toggleEmployee = (employeeId) => {
    setAssignmentError("");
    setAssignmentSuccess("");

    setSelectedEmployeeIds((previousIds) =>
      previousIds.includes(employeeId)
        ? previousIds.filter((id) => id !== employeeId)
        : [...previousIds, employeeId]
    );
  };

  const handleSaveAssignment = async () => {
    if (!selectedProject?.id) {
      setAssignmentError("No project is selected.");
      return;
    }

    setSavingAssignment(true);
    setAssignmentError("");
    setAssignmentSuccess("");

    try {
      await assignProjectEmployees(
        selectedProject.id,
        selectedEmployeeIds
      );

      const refreshedProjects = await fetchProjects();

      const updatedProject = refreshedProjects.find(
        (project) => String(project.id) === String(selectedProject.id)
      );

      if (updatedProject) {
        setSelectedProject(updatedProject);
      }

      setAssignmentSuccess("Employee assignments saved successfully.");
      setShowAssignmentModal(false);

      if (updatedProject) {
        if (assignmentReturnTo === "full") {
          setShowFullProjectModal(true);
          setShowProjectModal(false);
        } else {
          setShowProjectModal(true);
          setShowFullProjectModal(false);
        }
      } else {
        setShowProjectModal(false);
        setShowFullProjectModal(false);
        setSelectedProject(null);
      }
    } catch (err) {
      setAssignmentError(
        err?.message || "Failed to save employee assignments."
      );
    } finally {
      setSavingAssignment(false);
    }
  };

  // SHARED ASSIGNED RESOURCE LIST
  // ============================================================

  const renderAssignedResources = (project) => {

    return (

        <div className="project-section-heading">

            <h3>Assigned Resources</h3>
          </div>

          <strong>{team.length}</strong>
        </div>

        {team.length > 0 ? (
          <div className="project-team-list">
            {team.map((member, index) => {
              const memberName = getMemberName(member, index);
              const memberRole = getMemberRole(member);

              return (
                <div
                  className="project-team-member"
                  key={`${getMemberId(member) ?? memberName}-${index}`}
                >
                  <div className="project-team-avatar">
                    {memberName.charAt(0).toUpperCase()}
                  </div>

                  <div>
                    <strong>{memberName}</strong>
                    <span>{memberRole}</span>
                    {member?.email && <span>{member.email}</span>}
                  </div>
                </div>
              );
            })}
          </div>
        ) : (
          <p>No employees are assigned to this project yet.</p>
        )}

        <div style={{ marginTop: 16 }}>
          <button
            type="button"
            className="project-primary-btn"
            onClick={() =>
              openAssignmentModal(
                project,
                showFullProjectModal ? "full" : "details"
              )
            }
          >
            <UserPlus size={16} />
            Assign Employees
          </button>
        </div>
      </>
    );
  };

  // ============================================================
  // RENDER
  // ============================================================

  return (
    <div className="admin-projects">
      {/* PAGE HEADER */}

      <div className="admin-projects-header">
        <div className="admin-projects-header-content">
          <span className="admin-projects-eyebrow">
            PROJECT MANAGEMENT
          </span>

          <h1>Projects</h1>

          <p>
            Monitor projects, estimations, resources, progress, and
            project risks.
          </p>
        </div>
      </div>

      {/* SUMMARY CARDS */}

      <div className="admin-project-summary">
        <div className="project-summary-card">
          <div className="project-summary-icon">
            <FolderKanban size={19} />
          </div>
          <div>
            <span>Total Projects</span>
            <strong>{loading ? "—" : totalProjects}</strong>
          </div>
        </div>

        <div className="project-summary-card">
          <div className="project-summary-icon">
            <Activity size={19} />
          </div>
          <div>
            <span>Active Projects</span>
            <strong>{loading ? "—" : activeProjects}</strong>
          </div>
        </div>

        <div className="project-summary-card">
          <div className="project-summary-icon">
            <CheckCircle2 size={19} />
          </div>
          <div>
            <span>Completed</span>
            <strong>{loading ? "—" : completedProjects}</strong>
          </div>
        </div>

        <div className="project-summary-card">
          <div className="project-summary-icon">
            <AlertTriangle size={19} />
          </div>
          <div>
            <span>High Risk</span>
            <strong>{loading ? "—" : highRiskProjects}</strong>
          </div>
        </div>

                  {status}
                </option>

            </select>

            <select
              className="admin-project-filter"
              value={typeFilter}

              aria-label="Filter by project type"
            >
              <option value="All">All Types</option>

                <option key={type} value={type}>
                  {type}
                </option>
              ))}
            </select>

            <button
              type="button"
              className="project-action-btn"
              onClick={fetchProjects}
              disabled={loading}
              title="Refresh projects"
              aria-label="Refresh projects"
            >
              <RefreshCw
                size={16}
                className={loading ? "admin-projects-spin" : ""}
              />
            </button>
          </div>
        </div>

        {/* PROJECT TABLE */}

        {error && (
          <div className="admin-project-error" role="alert">
            <AlertTriangle size={17} />
            <span>{error}</span>
            <button type="button" onClick={fetchProjects}>

          </div>
        )}

        <div className="admin-project-table-wrapper">
          <table className="admin-project-table">

              <tr>

                <th>Owner</th>
                <th>Type</th>
                <th>Estimation</th>

                <th>Team</th>
                <th>Risk</th>
                <th>Status</th>
                <th>Actions</th>
              </tr>
            </thead>

            <tbody>
              {loading ? (
                <tr>
                  <td colSpan="9" className="project-empty-cell">
                    <RefreshCw
                      size={28}
                      className="admin-projects-spin"
                    />
                    <strong>Loading projects</strong>
                    <span>
                      Fetching project records from the backend.
                    </span>
                  </td>
                </tr>
              ) : currentProjects.length > 0 ? (
                currentProjects.map((project, index) => (
                  <tr
                    key={project.id}
                    className="admin-project-row"
                    style={{ "--row-index": index }}
                  >
                    <td>
                      <div className="project-name-cell">

                      <span className="project-type-text">
                        {project.type}

                      <div className="project-estimation-cell">
                        <strong>{project.cost}</strong>
                        <span>
                          <Clock3 size={13} />
                          {project.duration}
                        </span>
                      </div>
                    </td>

                    <td>
                      <div className="project-progress-cell">
                        <div className="project-progress-header">
                          <span>

                              ? "Not available"
                              : `${project.progress}%`}

                        </div>

                          <div
                            className="project-progress-fill"

                              width:
                                project.progress === null
                                  ? "0%"
                                  : `${project.progress}%`,
                            }}
                          />
                        </div>
                      </div>
                    </td>

                    <td>
                      <div className="project-team-cell">
                        <Users size={15} />
                        <span>{project.teamSize ?? "—"}</span>
                      </div>
                    </td>

                    <td>
                      {project.risk ? (
                        <span
                          className={`project-risk-badge ${getRiskClass(
                            project.risk
                          )}`}
                        >
                          {project.risk}
                        </span>
                      ) : (
                        <span>—</span>
                      )}
                    </td>

                    <td>
                      <span
                        className={`project-status-badge ${getStatusClass(
                          project.status
                        )}`}
                      >
                        {project.status}
                      </span>
                    </td>

                    <td>
                      <div className="project-actions">
                        <button
                          type="button"
                          className="project-action-btn"

                          onClick={() => openProject(project)}
                        >
                          <Eye size={16} />
                        </button>

                        <button
                          type="button"
                          className="project-action-btn"
                          title="View Full Project"
                          aria-label={`View full ${project.name}`}
                          onClick={() => openFullProject(project)}
                        >
                          <ClipboardList size={16} />
                        </button>
                      </div>
                    </td>
                  </tr>
                ))
              ) : (
                <tr>
                  <td colSpan="9" className="project-empty-cell">
                    <FolderKanban size={30} />
                    <strong>No projects found</strong>
                    <span>
                      Try changing your search or filters.
                    </span>
                  </td>
                </tr>
              )}
            </tbody>
          </table>
        </div>

        {/* PAGINATION */}

        <div className="projects-footer">
          <div className="projects-footer-info">
            Showing{" "}
            {filteredProjects.length === 0 ? 0 : startIndex + 1}
            {" - "}
            {Math.min(
              startIndex + projectsPerPage,
              filteredProjects.length
            )}
            {" of "}
            {filteredProjects.length} projects
          </div>

          <div className="projects-pagination">
            <button
              type="button"
              className="projects-page-btn"
              disabled={loading || safeCurrentPage === 1}
              onClick={() => changePage(safeCurrentPage - 1)}
            >
              <ChevronLeft size={15} />
              <span>Previous</span>
            </button>

            {Array.from(
              { length: totalPages },
              (_, index) => index + 1
            ).map((page) => (
              <button
                key={page}
                type="button"
                className={`projects-page-btn ${
                  safeCurrentPage === page ? "active" : ""
                }`}
                disabled={loading}
                onClick={() => changePage(page)}
              >
                {page}
              </button>
            ))}

            <button
              type="button"
              className="projects-page-btn"
              disabled={
                loading ||
                totalPages === 0 ||
                safeCurrentPage === totalPages
              }
              onClick={() => changePage(safeCurrentPage + 1)}
            >
              <span>Next</span>
              <ChevronRight size={15} />
            </button>
          </div>
        </div>
      </div>

      {/* PROJECT DETAILS MODAL */}

      {showProjectModal && selectedProject && (
        <div
          className="project-modal-overlay"
          onClick={closeProject}
        >
          <div
            className="project-modal"
            role="dialog"
            aria-modal="true"
            aria-labelledby="project-details-title"
            onClick={(event) => event.stopPropagation()}
          >
            <div className="project-modal-header">
              <div>
                <span className="project-modal-label">
                  PROJECT DETAILS
                </span>
                <h2 id="project-details-title">
                  {selectedProject.name}
                </h2>
              </div>

              <button
                type="button"
                className="project-modal-close"
                onClick={closeProject}
                aria-label="Close project details"
              >
                <X size={18} />
              </button>
            </div>

            <div className="project-modal-body">
              <div className="project-modal-status-row">
                <span
                  className={`project-status-badge ${getStatusClass(
                    selectedProject.status
                  )}`}
                >
                  {selectedProject.status}
                </span>

                {selectedProject.risk && (
                  <span
                    className={`project-risk-badge ${getRiskClass(
                      selectedProject.risk
                    )}`}
                  >
                    {selectedProject.risk} Risk
                  </span>
                )}
              </div>

              <div className="project-modal-description">
                <h3>Project Overview</h3>
                <p>{selectedProject.description}</p>
              </div>

              <div className="project-detail-grid">
                <div className="project-detail-item">
                  <span>Project Owner</span>
                  <strong>
                    <UserRound size={15} />
                    {selectedProject.owner}
                  </strong>
                </div>

                <div className="project-detail-item">
                  <span>Project Type</span>
                  <strong>{selectedProject.type}</strong>
                </div>

                <div className="project-detail-item">
                  <span>Estimated Cost</span>
                  <strong>
                    <DollarSign size={15} />
                    {selectedProject.cost}
                  </strong>
                </div>

                <div className="project-detail-item">
                  <span>Estimated Duration</span>
                  <strong>
                    <Clock3 size={15} />
                    {selectedProject.duration}
                  </strong>
                </div>

                <div className="project-detail-item">
                  <span>Created</span>
                  <strong>
                    <CalendarDays size={15} />
                    {selectedProject.created}
                  </strong>
                </div>

                <div className="project-detail-item">
                  <span>Deadline</span>
                  <strong>
                    <CalendarDays size={15} />
                    {selectedProject.deadline}
                  </strong>
                </div>
              </div>

              <div className="project-modal-section">
                <div className="project-section-heading">
                  <div>
                    <Activity size={17} />
                    <h3>Project Progress</h3>
                  </div>

                  <strong>
                    {selectedProject.progress === null
                      ? "Not available"
                      : `${selectedProject.progress}%`}
                  </strong>
                </div>

                <div className="project-progress-bar project-modal-progress">
                  <div
                    className="project-progress-fill"
                    style={{
                      width:
                        selectedProject.progress === null
                          ? "0%"
                          : `${selectedProject.progress}%`,
                    }}
                  />
                </div>
              </div>

              <div className="project-modal-section">
                {renderAssignedResources(selectedProject)}
              </div>
            </div>

            <div className="project-modal-actions">
              <button
                type="button"
                className="project-secondary-btn"
                onClick={closeProject}
              >
                Close
              </button>

              <button
                type="button"
                className="project-primary-btn"
                onClick={() => openFullProject(selectedProject)}
              >
                <ClipboardList size={16} />
                View Full Project
              </button>
            </div>
          </div>
        </div>
      )}

      {/* FULL PROJECT MODAL */}

      {showFullProjectModal && selectedProject && (
        <div
          className="project-modal-overlay"
          onClick={closeFullProject}
        >
          <div
            className="project-modal project-full-modal"
            role="dialog"
            aria-modal="true"
            aria-labelledby="full-project-title"
            onClick={(event) => event.stopPropagation()}
          >
            <div className="project-modal-header">
              <div>
                <span className="project-modal-label">
                  FULL PROJECT
                </span>
                <h2 id="full-project-title">
                  {selectedProject.name}
                </h2>
              </div>

              <button
                type="button"
                className="project-modal-close"
                onClick={closeFullProject}
                aria-label="Close full project"
              >
                <X size={18} />
              </button>
            </div>

            <div className="project-modal-body">
              <div className="project-modal-description">
                <h3>Project Overview</h3>
                <p>{selectedProject.description}</p>
              </div>

              <div className="project-modal-section">
                <div className="project-section-heading">
                  <div>
                    <FolderKanban size={17} />
                    <h3>Project Information</h3>
                  </div>
                </div>

                <div className="project-detail-grid">
                  <div className="project-detail-item">
                    <span>Project ID</span>
                    <strong>{displayValue(selectedProject.id)}</strong>
                  </div>

                  <div className="project-detail-item">
                    <span>Project Name</span>
                    <strong>{selectedProject.name}</strong>
                  </div>

                  <div className="project-detail-item">
                    <span>Owner</span>
                    <strong>{selectedProject.owner}</strong>
                  </div>

                  <div className="project-detail-item">
                    <span>Project Type</span>
                    <strong>{selectedProject.type}</strong>
                  </div>

                  <div className="project-detail-item">
                    <span>Status</span>
                    <strong>{selectedProject.status}</strong>
                  </div>

                  <div className="project-detail-item">
                    <span>Created</span>
                    <strong>{selectedProject.created}</strong>
                  </div>

                  <div className="project-detail-item">
                    <span>Deadline</span>
                    <strong>{selectedProject.deadline}</strong>
                  </div>
                </div>
              </div>

              <div className="project-modal-section">
                <div className="project-section-heading">
                  <div>
                    <DollarSign size={17} />
                    <h3>Estimation & Progress</h3>
                  </div>
                </div>

                <div className="project-detail-grid">
                  <div className="project-detail-item">
                    <span>Estimated Cost</span>
                    <strong>{selectedProject.cost}</strong>
                  </div>

                  <div className="project-detail-item">
                    <span>Estimated Duration</span>
                    <strong>{selectedProject.duration}</strong>
                  </div>

                  <div className="project-detail-item">
                    <span>Current Progress</span>
                    <strong>
                      {selectedProject.progress === null
                        ? "Not available"
                        : `${selectedProject.progress}%`}
                    </strong>
                  </div>

                  <div className="project-detail-item">
                    <span>Project Risk</span>
                    <strong>{displayValue(selectedProject.risk)}</strong>
                  </div>
                </div>

                <div className="project-progress-bar project-modal-progress">
                  <div
                    className="project-progress-fill"
                    style={{
                      width:
                        selectedProject.progress === null
                          ? "0%"
                          : `${selectedProject.progress}%`,
                    }}
                  />
                </div>
              </div>

              <div className="project-modal-section">
                {renderAssignedResources(selectedProject)}
              </div>

              <div className="project-modal-section">
                <div className="project-section-heading">
                  <div>
                    <ShieldAlert size={17} />
                    <h3>Project Record</h3>
                  </div>
                </div>

                <div className="project-admin-note">
                  <p>
                    Project information is displayed when the backend
                    provides it. Missing cost, duration, progress, risk,
                    deadline, and resource information is not fabricated.
                  </p>
                </div>
              </div>
            </div>

            <div className="project-modal-actions">
              <button
                type="button"
                className="project-primary-btn"
                onClick={closeFullProject}
              >
                Close
              </button>

              <button
                type="button"
                className="project-secondary-btn"

                          "project-assignment-option",

                          <span className="assignment-selected-indicator">

                <div className="assignment-empty-state">
                  <div className="assignment-empty-icon">
                    <Users size={25} />
                  </div>

                  <h3>No employees available</h3>

                  <p>
                    No employee records were returned. Try refreshing
                    the employee list.
                  </p>

                  <button
                    type="button"
                    className="project-secondary-btn"
                    onClick={fetchEmployees}
                    disabled={loadingEmployees || savingAssignment}
                  >
                    <RefreshCw size={15} />
                    Retry
                  </button>
                </div>
              )}
            </div>

            {/* Modal footer */}

            <div className="project-modal-actions assignment-modal-footer">
              <button
                type="button"
                className="project-secondary-btn"
                onClick={closeAssignmentModal}

              >
                {savingAssignment ? (
                  <>
                    <RefreshCw
                      size={16}
                      className="admin-projects-spin"
                    />
                    Saving...
                  </>
                ) : (
                  <>
                    <Save size={16} />
                    Save Assignments
                  </>
                )}
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

      projectEmployees.map((member) => Number(member.id))
    );

    const nonProjectMembers = selectedAssignments.filter(
      (employeeId) => !projectMemberIds.has(Number(employeeId))
    );

    if (nonProjectMembers.length > 0) {
      setAssignmentError(
        "Only employees assigned to the selected project can receive its tasks."
      );
      return;
    }

    const invalidMembers = selectedAssignments.filter((employeeId) => {
      const alreadyOnProject =
        (!isCreating &&
          selectedTask?.assignedTo.includes(Number(employeeId))) ||
        isMemberAlreadyOnProject(
          employeeId,
          Number(taskForm.projectId),
          !isCreating ? selectedTask?.id : null
        );

      if (alreadyOnProject) return false;

      return getMemberProjectCount(employeeId) >= MAX_PROJECTS_PER_MEMBER;
    });

    if (invalidMembers.length > 0) {
      const names = invalidMembers
        .map((id) => getEmployeeById(id)?.name || `Employee #${id}`)
        .join(", ");

      setAssignmentError(
        `${names} cannot be assigned because the active-project limit is ${MAX_PROJECTS_PER_MEMBER}.`
      );

      return;
    }

    setIsSaving(true);

    try {
      if (isCreating) {
        await createAdminTask({
          title: taskForm.title.trim(),
          description: taskForm.description.trim(),
          project_id: Number(taskForm.projectId),
          weekly_plan_id: Number(taskForm.weeklyPlanId),
          priority: toApiPriority(taskForm.priority),
          due_date: `${taskForm.dueDate}T00:00:00`,
          employee_ids: selectedAssignments.map(Number),
        });

        await loadPageData();

        setIsAssignOpen(false);
        setSelectedTask(null);
        setSelectedAssignments([]);
        setCurrentPage(1);
        setSuccessMessage("Task created and assigned successfully.");
      } else {
        if (!selectedTask) {
          setAssignmentError("No task is selected for updating.");
          return;
        }

        await updateAdminTask(selectedTask.id, {
          title: taskForm.title.trim(),
          description: taskForm.description.trim(),
          priority: toApiPriority(taskForm.priority),
          status: toApiStatus(taskForm.status),
          due_date: `${taskForm.dueDate}T00:00:00`,
          employee_ids: selectedAssignments.map(Number),
        });

        await loadPageData();

        setIsAssignOpen(false);
        setSelectedTask(null);
        setSelectedAssignments([]);
        setSuccessMessage("Task updated successfully.");
      }
    } catch (error) {
      setAssignmentError(
        getErrorMessage(
          error,
          isCreating
            ? "Unable to create the task."
            : "Unable to update the task."
        )
      );
    } finally {
      setIsSaving(false);
    }
  };

  const handleLoadAssignmentHistory = async () => {
    if (!selectedTask) return;

    setIsHistoryLoading(true);
    setHistoryError("");
    setAssignmentHistory([]);

    try {
      const result = await getAdminTaskAssignmentHistory(selectedTask.id);
      setAssignmentHistory(getArray(result));
    } catch (error) {
      setHistoryError(
        getErrorMessage(error, "Unable to load assignment history.")
      );
    } finally {
      setIsHistoryLoading(false);
    }
  };

  const handleCarryForward = async () => {
    if (!selectedTask) return;

    if (!carryForwardPlanId) {
      setHistoryError("Select a target weekly plan first.");
      return;
    }

    if (isCompletedStatus(selectedTask.status)) {
      setHistoryError(
        "Completed or cancelled tasks cannot be carried forward."
      );
      return;
    }

    const targetPlan = weeklyPlans.find(
      (plan) => Number(plan.id) === Number(carryForwardPlanId)
    );

    if (!targetPlan) {
      setHistoryError("The selected weekly plan could not be found.");
      return;
    }

    setIsCarryingForward(true);
    setHistoryError("");
    setSuccessMessage("");

    try {
      await carryForwardAdminTask(
        selectedTask.id,
        Number(carryForwardPlanId)
      );

      await loadPageData();
      await loadWeeklyPlans(selectedTask.projectId);

      setSelectedTask((current) =>
        current
          ? {
              ...current,
              weeklyPlanIds: Array.from(
                new Set([
                  ...(current.weeklyPlanIds || []),
                  Number(carryForwardPlanId),
                ])
              ),
            }
          : current
      );

      setCarryForwardPlanId("");
      setSuccessMessage("Task carried forward successfully.");
    } catch (error) {
      setHistoryError(
        getErrorMessage(error, "Unable to carry the task forward.")
      );
    } finally {
      setIsCarryingForward(false);
    }
  };

  const resetFilters = () => {
    setSearch("");
    setStatusFilter("All");
    setPriorityFilter("All");
    setProjectFilter("All");
    setCurrentPage(1);
  };

  return (
    <div className="admin-tasks-page">
      <div className="admin-tasks-header">
        <div className="admin-tasks-header-content">
          <div className="admin-tasks-eyebrow">Task Management</div>

          <h1 className="admin-tasks-main-title">
            Tasks &amp; Assignments
          </h1>

          <p>
            Manage project tasks, weekly plans, team assignments and employee
            workloads.
          </p>
        </div>

        <div className="admin-tasks-header-actions">
          <button
            type="button"
            className="admin-tasks-secondary-btn"
            onClick={loadPageData}
            disabled={isLoading}
          >
            <RefreshCcw size={15} />
            Refresh
          </button>

          <button
            type="button"
            className="admin-tasks-primary-btn"
            onClick={() => openAssignModal()}
          >
            <Plus size={17} />
            Create Task
          </button>
        </div>
      </div>

      {isLoading && (
        <div className="admin-tasks-notice" role="status">
          <RefreshCcw size={16} className="admin-tasks-loading-icon" />
          Loading tasks, projects and employees...
        </div>
      )}

      {successMessage && (
        <div className="admin-tasks-success" role="status">
          <CheckCircle2 size={16} />
          <span>{successMessage}</span>
          <button
            type="button"
            onClick={() => setSuccessMessage("")}
            aria-label="Dismiss message"
          >
            <X size={15} />
          </button>
        </div>
      )}

      <div className="admin-tasks-stats">
        {stats.map((stat, index) => {
          const Icon = stat.icon;

          return (
            <div
              key={stat.label}
              className="admin-task-stat-card"
              style={{ "--card-index": index }}
            >
              <div className="admin-task-stat-icon">
                <Icon size={19} />
              </div>
              <div className="admin-task-stat-content">
                <span>{stat.label}</span>
                <strong>{stat.value}</strong>
              </div>
            </div>
          );
        })}
      </div>

      <section className="admin-tasks-panel">
        <div className="admin-tasks-toolbar">
          <div className="admin-tasks-search">
            <Search size={17} />
            <input
              type="text"
              value={search}
              onChange={(event) => setSearch(event.target.value)}
              placeholder="Search tasks, projects or team members..."
              aria-label="Search tasks"
            />
          </div>

          <div className="admin-tasks-filters">
            <div className="admin-tasks-filter">
              <Filter
                size={15}
                className="admin-tasks-filter-icon"
                aria-hidden="true"
              />
              <select
                id="task-status-filter"
                aria-label="Filter by status"
                value={statusFilter}
                onChange={(event) => setStatusFilter(event.target.value)}
              >
                <option value="All">All Status</option>
                {TASK_STATUSES.map((status) => (
                  <option key={status} value={status}>
                    {status}
                  </option>
                ))}
              </select>
            </div>

            <div className="admin-tasks-filter">
              <Filter
                size={15}
                className="admin-tasks-filter-icon"
                aria-hidden="true"
              />
              <select
                id="task-priority-filter"
                aria-label="Filter by priority"
                value={priorityFilter}
                onChange={(event) => setPriorityFilter(event.target.value)}
              >
                <option value="All">All Priority</option>
                {TASK_PRIORITIES.map((priority) => (
                  <option key={priority} value={priority}>
                    {priority}
                  </option>
                ))}
              </select>
            </div>

            <div className="admin-tasks-filter">
              <Filter
                size={15}
                className="admin-tasks-filter-icon"
                aria-hidden="true"
              />
              <select
                id="task-project-filter"
                aria-label="Filter by project"
                value={projectFilter}
                onChange={(event) => setProjectFilter(event.target.value)}
              >
                <option value="All">All Projects</option>
                {projects.map((project) => (
                  <option key={project.id} value={String(project.id)}>
                    {project.name}
                  </option>
                ))}
              </select>
            </div>
          </div>
        </div>

        <div className="admin-tasks-table-wrap">
          {isLoading ? (
            <div className="admin-tasks-empty">
              <RefreshCcw size={25} className="admin-tasks-loading-icon" />
              <h3>Loading tasks</h3>
              <p>Fetching the latest task data from the backend.</p>
            </div>
          ) : paginatedTasks.length > 0 ? (
            <table className="admin-tasks-table">
              <thead>
                <tr>
                  <th>Task</th>
                  <th>Project</th>
                  <th>Assigned To</th>
                  <th>Priority</th>
                  <th>Due Date</th>
                  <th>Weekly Plans</th>
                  <th>Status</th>
                  <th>Actions</th>
                </tr>
              </thead>

              <tbody>
                {paginatedTasks.map((task, index) => {
                  const assignedMembers = task.assignedTo
                    .map(getEmployeeById)
                    .filter(Boolean);

                  return (
                    <tr
                      key={task.id}
                      className="admin-task-row"
                      style={{ "--row-index": index }}
                    >
                      <td>
                        <div className="admin-task-title-cell">
                          <div className="admin-task-title-icon">
                            <ClipboardList size={16} />
                          </div>
                          <div>
                            <strong>{task.title}</strong>
                            <span>
                              {task.assignedTo.length}{" "}
                              {task.assignedTo.length === 1
                                ? "member"
                                : "members"}{" "}
                              assigned
                            </span>
                          </div>
                        </div>
                      </td>

                      <td>
                        <div className="admin-task-project-cell">
                          <FolderKanban size={15} />
                          <span>{task.project}</span>
                        </div>
                      </td>

                      <td>
                        <div className="admin-task-assignees">
                          {assignedMembers.length > 0 ? (
                            assignedMembers.map((member, memberIndex) => (
                              <div
                                key={member.id}
                                className="admin-task-assignee"
                                style={{ "--member-index": memberIndex }}
                                title={`${member.name} — ${member.role}`}
                              >
                                <div className="admin-task-avatar">
                                  {getInitials(member.name)}
                                </div>
                                <div className="admin-task-assignee-info">
                                  <strong>{member.name}</strong>
                                  <span>{member.role}</span>
                                </div>
                              </div>
                            ))
                          ) : (
                            <span className="admin-task-unassigned">
                              Unassigned
                            </span>
                          )}
                        </div>
                      </td>

                      <td>
                        <span className={getPriorityClass(task.priority)}>
                          <Flag size={13} />
                          {task.priority}
                        </span>
                      </td>

                      <td>
                        <div
                          className={`admin-task-date ${
                            task.isOverdue ? "is-overdue" : ""
                          }`}
                        >
                          <CalendarDays size={14} />
                          <span>{formatDate(task.dueDate)}</span>
                          {task.isOverdue && (
                            <span
                              className="admin-task-overdue-label"
                              title="This task is overdue"
                            >
                              Overdue
                            </span>
                          )}
                        </div>
                      </td>

                      <td>
                        <span className="admin-task-weekly-plan-count">
                          {task.weeklyPlanIds.length}{" "}
                          {task.weeklyPlanIds.length === 1 ? "plan" : "plans"}
                        </span>
                      </td>

                      <td>
                        <span className={getStatusClass(task.status)}>
                          {task.status === "Completed" ? (
                            <CheckCircle2 size={13} />
                          ) : task.status === "Blocked" ? (
                            <AlertCircle size={13} />
                          ) : (
                            <Activity size={13} />
                          )}
                          {task.status}
                        </span>
                      </td>

                      <td>
                        <div className="admin-task-actions">
                          <button
                            type="button"
                            className="admin-task-action-btn"
                            onClick={() => openTask(task)}
                            title="View task details"
                            aria-label={`View ${task.title}`}
                          >
                            <Eye size={16} />
                          </button>

                          <button
                            type="button"
                            className="admin-task-action-btn admin-task-action-btn-secondary"
                            onClick={() => openAssignModal(task)}
                            title="Edit task and assignments"
                            aria-label={`Edit ${task.title}`}
                          >
                            <UserPlus size={16} />
                          </button>
                        </div>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          ) : (
            <div className="admin-tasks-empty">
              <div className="admin-tasks-empty-icon">
                <ClipboardList size={26} />
              </div>

              <h3>
                {loadError ? "Unable to load tasks" : "No tasks found"}
              </h3>

              <p>
                {loadError
                  ? "Tasks could not be loaded. Check the backend connection and try again."
                  : "Try changing your search or filters, or create a new task."}
              </p>

              {loadError ? (
                <button
                  type="button"
                  className="admin-tasks-secondary-btn"
                  onClick={loadPageData}
                >
                  <RefreshCcw size={15} />
                  Try Again
                </button>
              ) : (
                <button
                  type="button"
                  className="admin-tasks-secondary-btn"
                  onClick={resetFilters}
                >
                  <RefreshCcw size={15} />
                  Reset Filters
                </button>
              )}
            </div>
          )}
        </div>

        <div className="admin-tasks-footer">
          <span>
            {filteredTasks.length === 0
              ? "Showing 0 tasks"
              : `Showing ${pageStart + 1}–${Math.min(
                  pageStart + PAGE_SIZE,
                  filteredTasks.length
                )} of ${filteredTasks.length} tasks`}
          </span>

          <div className="admin-tasks-pagination">
            <button
              type="button"
              className="admin-task-page-btn"
              disabled={safeCurrentPage <= 1}
              onClick={() =>
                setCurrentPage((page) => Math.max(1, page - 1))
              }
            >
              <ChevronLeft size={15} />
              Previous
            </button>

            <div className="admin-task-page-number">
              Page <strong>{safeCurrentPage}</strong> of{" "}
              <strong>{totalPages}</strong>
            </div>

            <button
              type="button"
              className="admin-task-page-btn"
              disabled={safeCurrentPage >= totalPages}
              onClick={() =>
                setCurrentPage((page) => Math.min(totalPages, page + 1))
              }
            >
              Next
              <ChevronRight size={15} />
            </button>
          </div>
        </div>
      </section>

      {isDetailsOpen && selectedTask && (
        <div
          className="admin-task-modal-overlay"
          onMouseDown={closeTaskModal}
          role="presentation"
        >
          <div
            className="admin-task-modal admin-task-details-modal"
            onMouseDown={(event) => event.stopPropagation()}
            role="dialog"
            aria-modal="true"
            aria-labelledby="task-details-title"
          >
            <div className="admin-task-modal-header">
              <div>
                <span className="admin-task-modal-eyebrow">
                  Task Details
                </span>
                <h2 id="task-details-title">{selectedTask.title}</h2>
              </div>

              <button
                type="button"
                className="admin-task-modal-close"
                onClick={closeTaskModal}
                aria-label="Close task details"
              >
                <X size={18} />
              </button>
            </div>

            <div className="admin-task-detail-grid">
              <div className="admin-task-detail-card">
                <span>Project</span>
                <strong>{selectedTask.project}</strong>
              </div>

              <div className="admin-task-detail-card">
                <span>Due Date</span>
                <strong>{formatDate(selectedTask.dueDate)}</strong>
                {selectedTask.isOverdue && (
                  <span className="admin-task-overdue-label">Overdue</span>
                )}
              </div>

              <div className="admin-task-detail-card">
                <span>Priority</span>
                <strong>{selectedTask.priority}</strong>
              </div>

              <div className="admin-task-detail-card">
                <span>Status</span>
                <strong>
                  <span className={getStatusClass(selectedTask.status)}>
                    {selectedTask.status}
                  </span>
                </strong>
              </div>
            </div>

            <div className="admin-task-detail-section">
              <div className="admin-task-detail-section-title">
                <FileText size={16} />
                Description
              </div>
              <p>{selectedTask.description || "No description available."}</p>
            </div>

            <div className="admin-task-detail-section">
              <div className="admin-task-detail-section-title">
                <Users size={16} />
                Assigned Team Members
              </div>

              <div className="admin-task-detail-members">
                {selectedTask.assignedTo.length > 0 ? (
                  selectedTask.assignedTo.map((employeeId, index) => {
                    const member = getEmployeeById(employeeId);
                    const workload = getMemberProjectCount(employeeId);

                    return (
                      <div
                        key={employeeId}
                        className="admin-task-detail-member"
                        style={{ "--member-index": index }}
                      >
                        <div className="admin-task-avatar">
                          {getInitials(member?.name || `Employee ${employeeId}`)}
                        </div>

                        <div>
                          <strong>
                            {member?.name || `Employee #${employeeId}`}
                          </strong>
                          <span>{member?.role || "Team Member"}</span>
                        </div>

                        <small>
                          {workload}/{MAX_PROJECTS_PER_MEMBER} projects
                        </small>
                      </div>
                    );
                  })
                ) : (
                  <p>No employees are assigned to this task.</p>
                )}
              </div>
            </div>

            <div className="admin-task-detail-section">
              <div className="admin-task-detail-section-title">
                <CalendarDays size={16} />
                Weekly Plan History
              </div>

              <div className="admin-task-weekly-plan-history">
                {selectedTask.weeklyPlanIds.length > 0 ? (
                  selectedTask.weeklyPlanIds.map((planId) => {
                    const plan = weeklyPlans.find(
                      (item) => Number(item.id) === Number(planId)
                    );

                    return (
                      <div
                        key={planId}
                        className="admin-task-weekly-plan-item"
                      >
                        <strong>Weekly Plan #{planId}</strong>

                        {plan && (
                          <span>
                            {formatDate(plan.weekStartDate)} –{" "}
                            {formatDate(plan.weekEndDate)}
                          </span>
                        )}
                      </div>
                    );
                  })
                ) : (
                  <p>No weekly plan links were returned by the backend.</p>
                )}
              </div>
            </div>

            <div className="admin-task-detail-section admin-task-carry-forward">
              <div className="admin-task-detail-section-title">
                <ArrowRightLeft size={16} />
                Carry Forward Task
              </div>

              {isCompletedStatus(selectedTask.status) ? (
                <p>
                  Completed or cancelled tasks cannot be carried forward.
                </p>
              ) : (
                <>
                  <p>
                    Link this incomplete task to another weekly plan in the
                    same project. The original weekly plan remains unchanged.
                  </p>

                  <div className="admin-task-carry-forward-controls">
                    <select
                      aria-label="Target weekly plan"
                      value={carryForwardPlanId}
                      onChange={(event) =>
                        setCarryForwardPlanId(event.target.value)
                      }
                    >
                      <option value="">Select target weekly plan</option>

                      {weeklyPlans
                        .filter(
                          (plan) =>
                            Number(plan.id) !==
                            Number(selectedTask.weeklyPlanId)
                        )
                        .map((plan) => (
                          <option key={plan.id} value={String(plan.id)}>
                            Plan #{plan.id}: {formatDate(plan.weekStartDate)} –{" "}
                            {formatDate(plan.weekEndDate)}
                          </option>
                        ))}
                    </select>

                    <button
                      type="button"
                      className="admin-tasks-primary-btn"
                      onClick={handleCarryForward}
                      disabled={
                        isCarryingForward ||
                        !carryForwardPlanId ||
                        weeklyPlans.length < 2
                      }
                    >
                      <ArrowRightLeft size={15} />
                      {isCarryingForward
                        ? "Carrying Forward..."
                        : "Carry Forward"}
                    </button>
                  </div>

                  {weeklyPlans.length < 2 && (
                    <p>
                      No other weekly plan is available for this project yet.
                    </p>
                  )}
                </>
              )}
            </div>

            <div className="admin-task-detail-section">
              <div className="admin-task-detail-section-title">
                <History size={16} />
                Assignment History
              </div>

              <button
                type="button"
                className="admin-tasks-secondary-btn"
                onClick={handleLoadAssignmentHistory}
                disabled={isHistoryLoading}
              >
                <History size={15} />
                {isHistoryLoading
                  ? "Loading History..."
                  : "Load Assignment History"}
              </button>

              {historyError && (
                <div className="admin-task-assignment-error" role="alert">
                  <AlertCircle size={16} />
                  <span>{historyError}</span>
                </div>
              )}

              {assignmentHistory.length > 0 && (
                <div className="admin-task-history-list">
                  {assignmentHistory.map((entry, index) => {
                    const employee = getEmployeeById(entry.employee_id);

                    return (
                      <div
                        key={entry.id ?? `${entry.changed_at}-${index}`}
                        className="admin-task-history-item"
                      >
                        <div className="admin-task-history-icon">
                          <Clock3 size={15} />
                        </div>

                        <div className="admin-task-history-content">
                          <strong>
                            {capitalize(entry.action || "Assignment update")}
                          </strong>

                          <span>
                            Employee:{" "}
                            {employee?.name ||
                              (entry.employee_id
                                ? `Employee #${entry.employee_id}`
                                : "Not specified")}
                          </span>

                          <span>{formatDateTime(entry.changed_at)}</span>

                          {entry.reason && <p>{entry.reason}</p>}
                        </div>
                      </div>
                    );
                  })}
                </div>
              )}

              {!isHistoryLoading &&
                !historyError &&
                assignmentHistory.length === 0 && (
                  <p className="admin-task-history-empty">
                    Load the assignment history to view recorded changes.
                  </p>
                )}
            </div>

            <div className="admin-task-modal-footer">
              <button
                type="button"
                className="admin-tasks-secondary-btn admin-task-close-btn"
                onClick={closeTaskModal}
              >
                Close
              </button>

              <button
                type="button"
                className="admin-tasks-primary-btn"
                onClick={() => openAssignModal(selectedTask)}
              >
                <UserPlus size={15} />
                Edit Task
              </button>
            </div>
          </div>
        </div>
      )}

      {isAssignOpen && (
        <div
          className="admin-task-modal-overlay"
          onMouseDown={closeAssignModal}
          role="presentation"
        >
          <div
            className={`admin-task-modal admin-task-assignment-modal ${
              isCreating ? "admin-task-create-modal" : ""
            }`}
            onMouseDown={(event) => event.stopPropagation()}
            role="dialog"
            aria-modal="true"
            aria-labelledby="task-assignment-title"
          >
            <div className="admin-task-modal-header">
              <div>
                <span className="admin-task-modal-eyebrow">
                  {isCreating ? "Create & Assign" : "Task Management"}
                </span>

                <h2 id="task-assignment-title">
                  {isCreating ? "Create New Task" : "Edit Task"}
                </h2>
              </div>

              <button
                type="button"
                className="admin-task-modal-close"
                onClick={closeAssignModal}
                disabled={isSaving || isCreatingPlan}
                aria-label="Close task modal"
              >
                <X size={18} />
              </button>
            </div>

            <div className="admin-task-form-scroll">
              <div className="admin-task-form">
                <div className="admin-task-form-group">
                  <label htmlFor="task-title">
                    Task Name <span>*</span>
                  </label>

                  <input
                    id="task-title"
                    type="text"
                    value={taskForm.title}
                    onChange={(event) =>
                      setTaskForm((current) => ({
                        ...current,
                        title: event.target.value,
                      }))
                    }
                    placeholder="Enter task name"
                    required
                  />
                </div>

                <div className="admin-task-form-group">
                  <label htmlFor="task-project">
                    Project <span>*</span>
                  </label>

                  <select
                    id="task-project"
                    value={taskForm.projectId}
                    onChange={(event) =>
                      handleProjectChange(event.target.value)
                    }
                    disabled={!isCreating}
                    required
                  >
                    <option value="">Select a project</option>

                    {projects.map((project) => (
                      <option key={project.id} value={String(project.id)}>
                        {project.name}
                        {isInactiveProject(project.status)
                          ? " (Inactive)"
                          : ""}
                      </option>
                    ))}
                  </select>

                  {!isCreating && (
                    <small>
                      The original project cannot be changed through task
                      update.
                    </small>
                  )}
                </div>

                {isCreating && (
                  <div className="admin-task-form-group">
                    <label htmlFor="task-weekly-plan">
                      Weekly Plan <span>*</span>
                    </label>

                    <select
                      id="task-weekly-plan"
                      value={taskForm.weeklyPlanId}
                      onChange={(event) =>
                        setTaskForm((current) => ({
                          ...current,
                          weeklyPlanId: event.target.value,
                        }))
                      }
                      disabled={!taskForm.projectId}
                      required
                    >
                      <option value="">Select a weekly plan</option>

                      {weeklyPlans.map((plan) => (
                        <option key={plan.id} value={String(plan.id)}>
                          Plan #{plan.id}: {formatDate(plan.weekStartDate)} –{" "}
                          {formatDate(plan.weekEndDate)}
                        </option>
                      ))}
                    </select>

                    {taskForm.projectId && (
                      <div className="admin-task-weekly-plan-actions">
                        <span>
                          {weeklyPlans.length === 0
                            ? "No weekly plans found for this project."
                            : `${weeklyPlans.length} weekly plan(s) available.`}
                        </span>

                        <button
                          type="button"
                          className="admin-tasks-secondary-btn"
                          onClick={() => {
                            setAssignmentError("");
                            setIsPlanFormOpen((current) => !current);
                          }}
                        >
                          <Plus size={14} />
                          {isPlanFormOpen
                            ? "Cancel New Plan"
                            : "Create Weekly Plan"}
                        </button>
                      </div>
                    )}

                    {isPlanFormOpen && (
                      <div className="admin-task-weekly-plan-form">
                        <div className="admin-task-form-row">
                          <div className="admin-task-form-group">
                            <label htmlFor="weekly-plan-start">
                              Week Start <span>*</span>
                            </label>

                            <input
                              id="weekly-plan-start"
                              type="date"
                              value={planForm.weekStartDate}
                              onChange={(event) =>
                                setPlanForm((current) => ({
                                  ...current,
                                  weekStartDate: event.target.value,
                                }))
                              }
                            />
                          </div>

                          <div className="admin-task-form-group">
                            <label htmlFor="weekly-plan-end">
                              Week End <span>*</span>
                            </label>

                            <input
                              id="weekly-plan-end"
                              type="date"
                              value={planForm.weekEndDate}
                              min={planForm.weekStartDate || undefined}
                              onChange={(event) =>
                                setPlanForm((current) => ({
                                  ...current,
                                  weekEndDate: event.target.value,
                                }))
                              }
                            />
                          </div>
                        </div>

                        <button
                          type="button"
                          className="admin-tasks-primary-btn"
                          onClick={handleCreateWeeklyPlan}
                          disabled={isCreatingPlan}
                        >
                          <Save size={15} />
                          {isCreatingPlan
                            ? "Creating Weekly Plan..."
                            : "Save Weekly Plan"}
                        </button>
                      </div>
                    )}
                  </div>
                )}

                {!isCreating && (
                  <div className="admin-task-form-group">
                    <label>Original Weekly Plan</label>

                    <input
                      type="text"
                      value={
                        selectedTask?.weeklyPlanId
                          ? `Weekly Plan #${selectedTask.weeklyPlanId}`
                          : "No weekly plan"
                      }
                      readOnly
                    />

                    <small>
                      Use Carry Forward from task details to link this task to
                      another weekly plan.
                    </small>
                  </div>
                )}

                <div className="admin-task-form-row">
                  <div className="admin-task-form-group">
                    <label htmlFor="task-priority">Priority</label>

                    <select
                      id="task-priority"
                      value={taskForm.priority}
                      onChange={(event) =>
                        setTaskForm((current) => ({
                          ...current,
                          priority: event.target.value,
                        }))
                      }
                    >
                      {TASK_PRIORITIES.map((priority) => (
                        <option key={priority} value={priority}>
                          {priority}
                        </option>
                      ))}
                    </select>
                  </div>

                  <div className="admin-task-form-group">
                    <label htmlFor="task-due-date">
                      Due Date <span>*</span>
                    </label>

                    <input
                      id="task-due-date"
                      type="date"
                      value={taskForm.dueDate}
                      onChange={(event) =>
                        setTaskForm((current) => ({
                          ...current,
                          dueDate: event.target.value,
                        }))
                      }
                      required
                    />
                  </div>
                </div>

                {!isCreating && (
                  <div className="admin-task-form-group">
                    <label htmlFor="task-status">Task Status</label>

                    <select
                      id="task-status"
                      value={taskForm.status}
                      onChange={(event) =>
                        setTaskForm((current) => ({
                          ...current,
                          status: event.target.value,
                        }))
                      }
                    >
                      {TASK_STATUSES.map((status) => (
                        <option key={status} value={status}>
                          {status}
                        </option>
                      ))}
                    </select>
                  </div>
                )}

                <div className="admin-task-form-group">
                  <label htmlFor="task-description">
                    Description <span>*</span>
                  </label>

                  <textarea
                    id="task-description"
                    rows="3"
                    value={taskForm.description}
                    onChange={(event) =>
                      setTaskForm((current) => ({
                        ...current,
                        description: event.target.value,
                      }))
                    }
                    placeholder="Describe the task..."
                    required
                  />
                </div>

                <div className="admin-task-assignment-section">
                  <div className="admin-task-assignment-heading">
                    <div>
                      <label>
                        Assign Team Members <span>*</span>
                      </label>

                      <p>
                        {taskForm.projectId
                          ? "Only employees assigned to this project are shown."
                          : "Select a project first to view its assigned employees."}
                      </p>
                    </div>

                    <div className="admin-task-assignment-count">
                      {selectedAssignments.length} selected
                    </div>
                  </div>

                  <div className="admin-task-member-list">
                    {!taskForm.projectId ? (
                      <div className="admin-tasks-empty">
                        <FolderKanban size={23} />
                        <h3>Select a project</h3>
                        <p>
                          Choose a project above to see its assigned employees.
                        </p>
                      </div>
                    ) : projectEmployees.length > 0 ? (
                      projectEmployees.map((member, index) => {
                        const isSelected = selectedAssignments.includes(
                          Number(member.id)
                        );

                        const projectCount = getMemberProjectCount(member.id);
                        const isLimitReached =
                          !canAssignMember(member.id) && !isSelected;

                        return (
                          <button
                            type="button"
                            key={member.id}
                            className={`admin-task-member-option ${
                              isSelected ? "is-selected" : ""
                            } ${isLimitReached ? "is-disabled" : ""}`}
                            style={{ "--member-index": index }}
                            onClick={() => handleAssignmentChange(member.id)}
                            disabled={isLimitReached || isSaving}
                            title={
                              isLimitReached
                                ? "Maximum active project limit reached"
                                : member.email || member.name
                            }
                          >
                            <div className="admin-task-member-option-main">
                              <div className="admin-task-avatar">
                                {getInitials(member.name)}
                              </div>

                              <div className="admin-task-member-option-info">
                                <strong>{member.name}</strong>
                                <span>{member.role}</span>
                              </div>
                            </div>

                            <div className="admin-task-member-workload">
                              <strong>
                                {projectCount}/{MAX_PROJECTS_PER_MEMBER}
                              </strong>

                              <span>
                                {isLimitReached
                                  ? "Limit Reached"
                                  : "Active Projects"}
                              </span>
                            </div>

                            <div className="admin-task-member-check">
                              {isSelected ? (
                                <CheckCircle2 size={19} />
                              ) : (
                                <span />
                              )}
                            </div>
                          </button>
                        );
                      })
                    ) : (
                      <div className="admin-tasks-empty">
                        <Users size={23} />
                        <h3>No project employees found</h3>
                        <p>
                          This project has no employees returned by its team
                          data or existing task assignments. Check the project
                          team assignments in the backend.
                        </p>
                      </div>
                    )}
                  </div>
                </div>

                {assignmentError && (
                  <div className="admin-task-assignment-error" role="alert">
                    <AlertCircle size={17} />
                    <span>{assignmentError}</span>
                  </div>
                )}
              </div>
            </div>

            <div className="admin-task-modal-footer admin-task-assignment-footer">
              <button
                type="button"
                className="admin-tasks-secondary-btn admin-task-close-btn"
                onClick={closeAssignModal}
                disabled={isSaving || isCreatingPlan}
              >
                Close
              </button>

              <button
                type="button"
                className="admin-tasks-primary-btn"
                onClick={handleSaveAssignment}
                disabled={isSaving || isCreatingPlan}
              >
                <Save size={15} />
                {isSaving
                  ? "Saving..."
                  : isCreating
                    ? "Create & Assign"
                    : "Save Changes"}
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

export default AdminTasks;
